# UFW Firewall Configuration for NexaForge Security Layer (L11)

# Default policies
*filter
:ufw-user-input - [0:0]
:ufw-user-output - [0:0]
:ufw-user-forward - [0:0]
:ufw-before-logs - [0:0]
:ufw-user-logs - [0:0]
:ufw-after-logs - [0:0]

# Allow loopback
-A ufw-user-input -i lo -j ACCEPT
-A ufw-user-output -o lo -j ACCEPT

# Allow established connections
-A ufw-user-input -m state --state RELATED,ESTABLISHED -j ACCEPT

# Allow services
-A ufw-user-input -p tcp --dport 22 -j ACCEPT                      # SSH
-A ufw-user-input -p tcp --dport 80 -j ACCEPT                      # HTTP
-A ufw-user-input -p tcp --dport 443 -j ACCEPT                     # HTTPS
-A ufw-user-input -p tcp --dport 8000 -j ACCEPT                    # API Gateway
-A ufw-user-input -p tcp --dport 8080 -j ACCEPT                    # Additional Service
-A ufw-user-input -p tcp --dport 5432 -j ACCEPT                    # PostgreSQL
-A ufw-user-input -p tcp --dport 27017 -j ACCEPT                   # MongoDB
-A ufw-user-input -p tcp --dport 6379 -j ACCEPT                    # Redis
-A ufw-user-input -p tcp --dport 9090 -j ACCEPT                    # Prometheus
-A ufw-user-input -p tcp --dport 3000 -j ACCEPT                    # Grafana
-A ufw-user-input -p tcp --dport 3100 -j ACCEPT                    # Loki
-A ufw-user-input -p tcp --dport 9093 -j ACCEPT                    # Alertmanager

# Deny all other incoming connections by default
-A ufw-user-input -j DROP

# Allow all outgoing connections by default
-A ufw-user-output -j ACCEPT

# Forward policy
-A ufw-user-forward -j DROP

COMMIT

# Custom rules for NexaForge
# Rate limiting for API endpoints
-A ufw-user-input -p tcp --dport 8000 -m limit --limit 100/minute -j ACCEPT
-A ufw-user-input -p tcp --dport 8000 -j DROP

# Block known malicious IPs (example)
# -A ufw-user-input -s 192.168.100.10 -j DROP

# Log suspicious activity
-A ufw-before-logs -m limit --limit 10/min -j LOG --log-prefix "[UFW BLOCK] "
